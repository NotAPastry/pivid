#-------RPi camera module-------
# libcamera/rpicam-vid camera stack. Auto-detect is OFF because the IMX290
# is not in the libcamera auto-detect list — it's bound explicitly via
# `dtoverlay=imx290,...` below. Do NOT also uncomment `start_x=1` — that
# enables the legacy MMAL stack which is mutually exclusive.
camera_auto_detect=0
#disable_camera_led=1

#-------GPU memory splits-------
# gpu_mem_512=76 (not the default 16) is the minimum that makes the
# firmware loader pick start.elf instead of start_cd.elf on the Pi
# Zero 2W. start_cd.elf is the cut-down VPU firmware that explicitly
# omits camera + codec services -- without them, libcamera reports
# "No cameras available!" even with the camera ribbon attached, and
# `modprobe bcm2835-unicam` binds to nothing because no CSI sensor
# can be brought up. The cma= reservation in cmdline.txt is
# independent and additional.
# 60 MB extra reservation; leaves ~324 MB ARM RAM, well above the
# daemon + rpicam-vid working set.
gpu_mem_256=16
gpu_mem_512=76
gpu_mem_1024=16

#-------Boot splash screen------
disable_splash=1

#-------Serial/UART-----
# PL011 on GPIO14/15 is reserved for the flight controller (not a serial console).
# Combined with dtoverlay=disable-bt below, GPIO14/15 = full PL011 (FIFO, stable
# baud rate independent of core_freq), not the mini-UART.
# The "core_freq=250" warning in the upstream docs only applies when BT is on the UART.
enable_uart=1

#-------SD card HPD-----
# Comment to enable SD card hot-plug detection, while booting via USB or network.
# NB: This causes constant CPU load and kernel errors when no SD card is inserted.
dtparam=sd_poll_once

arm_64bit=1

# No vc4-kms-v3d on TX: headless, no display driver needed. libcamera
# accesses the ISP/V4L2 camera subsystem directly without it.

# Detach Bluetooth modem from the PL011 UART:
# - Frees GPIO14/15 PL011 for flight-controller comms.
# - Disables the BT radio entirely.
# - Removes BT init wait from boot (~3 s).
# Userspace bluetooth.service and hciuart.service are masked in
# Automation_Custom_Script.sh so they don't error on boot.
dtoverlay=disable-bt

# IMX290 CSI sensor with 37.125 MHz external clock (sensor datasheet
# default; matches the on-board oscillator on our camera module).
dtoverlay=imx290,clock-frequency=37125000
